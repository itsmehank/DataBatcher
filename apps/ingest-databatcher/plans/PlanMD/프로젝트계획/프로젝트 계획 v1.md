### 요약 평가
설계 방향이 매우 좋습니다. OOP로 수집 대상을 추상화하고, 초기 적재/일일 갱신/갭 보충을 분리한 점, 그리고 `pandas_market_calendars`·SQLAlchemy·멀티스레딩을 고려한 점이 실무 요건에 부합합니다. 다만, 운영 안정성과 데이터 정합성, 재처리(리트라이/업서트/멱등성), 시간대/캘린더 처리, 스키마 최적화, 관측 가능성(로그·메트릭) 측면에서 몇 가지 필수 보완이 필요합니다.

### core/date_utils.py (예시 구현)
```python
# core/date_utils.py
from __future__ import annotations
import datetime as dt
import pytz
import pandas_market_calendars as mcal

class DateUtils:
    def __init__(self, tz_name: str = "Asia/Seoul"):
        self.tz = pytz.timezone(tz_name)
        self._calendars = {
            "XKRX": mcal.get_calendar("XKRX"),
            "NYSE": mcal.get_calendar("XNYS"),
        }

    def now(self) -> dt.datetime:
        return dt.datetime.now(self.tz)

    def is_weekend(self, date: dt.date) -> bool:
        return date.weekday() >= 5

    def is_market_open(self, market_code: str, date: dt.date) -> bool:
        cal = self._calendars[market_code]
        schedule = cal.schedule(start_date=date, end_date=date)
        return not schedule.empty

    def get_last_business_day(self, market_code: str, ref: dt.date | None = None) -> dt.date:
        ref = ref or self.now().date()
        cal = self._calendars[market_code]
        schedule = cal.schedule(start_date=ref - dt.timedelta(days=10), end_date=ref)
        return schedule.index.max().date()

    def after_market_close_with_buffer(self, market_code: str, local_close_hhmm: str, buffer_minutes: int = 30) -> bool:
        now = self.now()
        hh, mm = map(int, local_close_hhmm.split(":"))
        close_dt = now.replace(hour=hh, minute=mm, second=0, microsecond=0)
        close_dt = close_dt + dt.timedelta(minutes=buffer_minutes)
        return now >= close_dt
```

### core/db_manager.py (싱글톤 + 업서트 유틸)
```python
# core/db_manager.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Optional
from sqlalchemy import create_engine, text
from sqlalchemy.engine import Engine

@dataclass
class DBConfig:
    url: str
    pool_size: int = 10
    max_overflow: int = 20
    pool_recycle: int = 1800
    pool_pre_ping: bool = True

class DBManager:
    _engine: Optional[Engine] = None

    @classmethod
    def get_engine(cls, cfg: DBConfig) -> Engine:
        if cls._engine is None:
            cls._engine = create_engine(
                cfg.url,
                pool_size=cfg.pool_size,
                max_overflow=cfg.max_overflow,
                pool_recycle=cfg.pool_recycle,
                pool_pre_ping=cfg.pool_pre_ping,
                future=True,
            )
        return cls._engine

    @classmethod
    def get_latest_date(cls, engine: Engine, table: str, symbol: str) -> Optional[str]:
        sql = text(f"SELECT MAX(`date`) AS max_date FROM `{table}` WHERE `symbol`=:symbol")
        with engine.connect() as conn:
            row = conn.execute(sql, {"symbol": symbol}).mappings().first()
            return row["max_date"].isoformat() if row and row["max_date"] else None

    @classmethod
    def upsert_dataframe(cls, engine: Engine, df, table: str, unique_keys=("symbol","date")):
        # df: pandas DataFrame with columns (= table columns)
        if df.empty:
            return 0
        cols = list(df.columns)
        placeholders = ",".join(["%s"] * len(cols))
        cols_escaped = ",".join([f"`{c}`" for c in cols])
        update_clause = ",".join([f"`{c}`=VALUES(`{c}`)" for c in cols if c not in unique_keys])
        sql = f"""
        INSERT INTO `{table}` ({cols_escaped}) VALUES ({placeholders})
        ON DUPLICATE KEY UPDATE {update_clause}
        """
        data = [tuple(row[c] for c in cols) for _, row in df.iterrows()]
        with engine.begin() as conn:
            conn.exec_driver_sql(sql, data)
        return len(data)
```

### core/base_collector.py (추상 클래스)
```python
# core/base_collector.py
from __future__ import annotations
from abc import ABC, abstractmethod
import pandas as pd

class BaseCollector(ABC):
    category: str

    @abstractmethod
    def fetch(self, symbol: str, start, end) -> pd.DataFrame:
        ...

    def validate(self, df: pd.DataFrame) -> pd.DataFrame:
        if df is None or df.empty:
            return df
        # 기본 검증 규칙: 음수 거래량/0 가격 제거
        mask = (
            (df["Open"].fillna(0) > 0)
            & (df["High"].fillna(0) > 0)
            & (df["Low"].fillna(0) > 0)
            & (df["Close"].fillna(0) > 0)
            & (df["Volume"].fillna(0) >= 0)
        )
        return df.loc[mask]

    @abstractmethod
    def save(self, df: pd.DataFrame, symbol: str) -> int:
        ...
```

### collectors/kr_stock.py (FDR 연동 예시)
```python
# collectors/kr_stock.py
from __future__ import annotations
import FinanceDataReader as fdr
import pandas as pd
from core.base_collector import BaseCollector
from core.db_manager import DBManager

class KRStockCollector(BaseCollector):
    category = "stock"

    def __init__(self, engine, table="stock_prices"):
        self.engine = engine
        self.table = table

    def fetch(self, symbol: str, start, end) -> pd.DataFrame:
        df = fdr.DataReader(symbol, start=start, end=end)  # index: Date
        if df is None or df.empty:
            return df
        df = df.reset_index().rename(columns=str.lower)  # date/open/high/low/close/volume/adj close
        df["symbol"] = symbol
        df["market"] = "XKRX"
        df["source"] = "FDR"
        # 컬럼 정렬
        cols = ["symbol","date","open","high","low","close","adj close","volume","market","source"]
        df = df[cols]
        df = df.rename(columns={"adj close": "adj_close"})
        # date를 DATE 타입으로 정규화
        df["date"] = pd.to_datetime(df["date"]).dt.date
        return df

    def save(self, df: pd.DataFrame, symbol: str) -> int:
        return DBManager.upsert_dataframe(self.engine, df, table=self.table)
```

### scripts/daily_update.py (실행 흐름 예시)
```python
# scripts/daily_update.py
from __future__ import annotations
import yaml
from core.date_utils import DateUtils
from core.db_manager import DBManager, DBConfig
from collectors.kr_stock import KRStockCollector

def main():
    cfg = yaml.safe_load(open("config/settings.yaml"))
    db_cfg = DBConfig(**cfg["database"])
    engine = DBManager.get_engine(db_cfg)

    du = DateUtils(cfg["runtime"]["timezone"])

    # 한국장 마감 체크 + 버퍼
    if not du.after_market_close_with_buffer("XKRX", cfg["markets"]["close_times"]["XKRX"], cfg["markets"]["safe_delay_minutes"]):
        print("Not time yet. Exiting.")
        return

    symbols = ["005930", "000660"]  # 예시: 설정/DB/파일에서 로드
    collector = KRStockCollector(engine)

    # 최신 날짜 조회 후 오늘 1일치만
    last_biz = du.get_last_business_day("XKRX")
    for s in symbols:
        latest = DBManager.get_latest_date(engine, collector.table, s)
        start = (latest or last_biz)
        df = collector.fetch(s, start=start, end=last_biz)
        df = collector.validate(df)
        if df is not None and not df.empty:
            collector.save(df, s)

if __name__ == "__main__":
    main()
```

### 멀티스레딩 + 레이트 리밋 + 리트라이 패턴
```python
from concurrent.futures import ThreadPoolExecutor, as_completed
import time
import random

def rate_limited_map(func, items, rate_per_sec=5, max_workers=8):
    interval = 1.0 / max(1, rate_per_sec)
    results = []
    with ThreadPoolExecutor(max_workers=max_workers) as ex:
        futures = []
        last = 0.0
        for item in items:
            now = time.time()
            sleep = max(0.0, interval - (now - last))
            if sleep: time.sleep(sleep)
            last = time.time()
            futures.append(ex.submit(func, item))
        for f in as_completed(futures):
            results.append(f.result())
    return results

# 리트라이 예시(간단)
def with_retry(fn, max_attempts=5, backoff_base=1, backoff_max=30):
    attempt = 0
    while True:
        try:
            return fn()
        except Exception as e:
            attempt += 1
            if attempt >= max_attempts:
                raise
            delay = min(backoff_max, backoff_base * (2 ** (attempt - 1)))
            delay *= random.uniform(0.8, 1.2)
            time.sleep(delay)
```

### 데이터 검증 규칙(확장)
- 가격(Open/High/Low/Close) 모두 양수, `Low <= High`, `Open`·`Close`는 `High`/`Low` 범위 내
- 거래량은 0 이상, 비정상 급증(상위 99.9 분위수 초과)은 경고 로깅
- 동일 `(symbol, date)` 중복 시 덮어쓰기(업서트), 하루치에 결측 컬럼이 많으면 스킵 또는 보류 큐로 이동
