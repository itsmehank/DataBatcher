### 목표
- 기존 수집 파이프라인(Init/Daily/Gap) 설계를 유지하면서, 이동평균선, RSI 등 다양한 지표를 “모듈형(플러그인)”으로 쉽게 추가/조합/교체할 수 있는 구조를 제안합니다.
- 지표는 저장 전략을 선택 가능: 1) 가격 테이블에 컬럼으로 병합(materialize in wide form), 2) 전용 지표 테이블에 세로형(long form)으로 저장, 3) 저장하지 않고 요청 시 계산(on-demand) — 프로젝트/성능 요구에 맞게 하이브리드 운용.

### DB 스키마 설계(지표 전용 테이블: long form)
```sql
CREATE TABLE IF NOT EXISTS stock_indicators (
  symbol        VARCHAR(32) NOT NULL,
  date          DATE        NOT NULL,
  indicator     VARCHAR(64) NOT NULL,     -- 예: sma, rsi, macd_signal
  params_hash   CHAR(40)    NOT NULL,     -- 파라미터 해시(sha1 등)
  value         DECIMAL(28,10) NULL,
  market        VARCHAR(16) NOT NULL,     -- XKRX/NYSE 등
  source        VARCHAR(16) NOT NULL,     -- 파생 원천(FDR)
  etl_loaded_at TIMESTAMP   NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (symbol, date, indicator, params_hash),
  KEY idx_indicator_date (indicator, date),
  KEY idx_symbol_date (symbol, date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```
- 파라미터가 다른 동일 지표를 동시 저장 가능(예: `sma_5`, `sma_20`).
- `params_hash`는 지표 파라미터를 고유 식별(버전 관리/멱등성)하고, 업서트에 활용.

Wide form 확장(선택):
- 가격 테이블에 `ind_sma_5`, `ind_rsi_14` 컬럼 추가. 지표 추가 시 마이그레이션 필요. 성능 중시/지표 셋 고정 시 유효.

### Indicator 추상화와 레지스트리
```python
# indicators/base_indicator.py
from __future__ import annotations
from abc import ABC, abstractmethod
import pandas as pd
from dataclasses import dataclass
from typing import Dict, Any, Optional

@dataclass(frozen=True)
class IndicatorParams:
    name: str              # "sma", "rsi" 등
    params: Dict[str, Any]

class BaseIndicator(ABC):
    name: str  # 레지스트리 키

    @abstractmethod
    def required_columns(self, p: IndicatorParams) -> list[str]:
        ...

    @abstractmethod
    def warmup(self, p: IndicatorParams) -> int:
        """증분 계산 시 최소 선행 필요 일수(rolling window)."""
        ...

    @abstractmethod
    def compute(self, df: pd.DataFrame, p: IndicatorParams) -> pd.Series:
        """index: date, 반환 Series.name은 식별 가능하게 설정."""
        ...
```

```python
# indicators/registry.py
from __future__ import annotations
from typing import Dict, Type
from .base_indicator import BaseIndicator

class IndicatorRegistry:
    _reg: Dict[str, Type[BaseIndicator]] = {}

    @classmethod
    def register(cls, ind_cls: Type[BaseIndicator]):
        cls._reg[ind_cls.name] = ind_cls
        return ind_cls

    @classmethod
    def get(cls, name: str) -> Type[BaseIndicator]:
        if name not in cls._reg:
            raise KeyError(f"Indicator not found: {name}")
        return cls._reg[name]

    @classmethod
    def list(cls):
        return list(cls._reg.keys())
```

# indicators/common/rsi.py
import pandas as pd
from indicators.base_indicator import BaseIndicator, IndicatorParams
from indicators.registry import IndicatorRegistry

@IndicatorRegistry.register
class RSI(BaseIndicator):
    name = "rsi"

    def required_columns(self, p):
        return [p.params.get("column", "close")]

    def warmup(self, p):
        return int(p.params.get("window", 14)) * 3

    def compute(self, df: pd.DataFrame, p):
        col = p.params.get("column", "close")
        win = int(p.params.get("window", 14))
        delta = df[col].diff()
        up = delta.clip(lower=0).ewm(alpha=1/win, adjust=False).mean()
        down = (-delta.clip(upper=0)).ewm(alpha=1/win, adjust=False).mean()
        rs = up / down.replace(0, pd.NA)
        rsi = 100 - (100 / (1 + rs))
        rsi.name = f"rsi_{win}_{col}"
        return rsi
```

# indicators/common/macd.py
import pandas as pd
from indicators.base_indicator import BaseIndicator, IndicatorParams
from indicators.registry import IndicatorRegistry

@IndicatorRegistry.register
class MACD(BaseIndicator):
    name = "macd"

    def required_columns(self, p):
        return [p.params.get("column", "close")]

    def warmup(self, p):
        fast = int(p.params.get("fast", 12))
        slow = int(p.params.get("slow", 26))
        signal = int(p.params.get("signal", 9))
        return (slow * 3) + signal

    def compute(self, df: pd.DataFrame, p):
        col = p.params.get("column", "close")
        fast = int(p.params.get("fast", 12))
        slow = int(p.params.get("slow", 26))
        signal = int(p.params.get("signal", 9))
        ema_fast = df[col].ewm(span=fast, adjust=False).mean()
        ema_slow = df[col].ewm(span=slow, adjust=False).mean()
        macd = ema_fast - ema_slow
        macd_signal = macd.ewm(span=signal, adjust=False).mean()
        macd_hist = macd - macd_signal
        # 반환을 Series 하나로 표준화할지 여부: long form 저장이면 3개를 각각 저장
        # pipeline에서 다중 시리즈 지원
        return {
            f"macd_{fast}_{slow}_{signal}_{col}": macd,
            f"macd_signal_{fast}_{slow}_{signal}_{col}": macd_signal,
            f"macd_hist_{fast}_{slow}_{signal}_{col}": macd_hist,
        }
```

# indicators/common/bollinger.py
import pandas as pd
from indicators.base_indicator import BaseIndicator, IndicatorParams
from indicators.registry import IndicatorRegistry

@IndicatorRegistry.register
class Bollinger(BaseIndicator):
    name = "bollinger"

    def required_columns(self, p):
        return [p.params.get("column", "close")]

    def warmup(self, p):
        return int(p.params.get("window", 20))

    def compute(self, df: pd.DataFrame, p):
        col = p.params.get("column", "close")
        win = int(p.params.get("window", 20))
        k = float(p.params.get("std", 2.0))
        ma = df[col].rolling(win, min_periods=win).mean()
        sd = df[col].rolling(win, min_periods=win).std(ddof=0)
        upper = ma + k * sd
        lower = ma - k * sd
        return {
            f"bb_mid_{win}_{k}_{col}": ma,
            f"bb_up_{win}_{k}_{col}": upper,
            f"bb_low_{win}_{k}_{col}": lower,
        }

### 파라미터 해시/멱등성 유틸
```python
# core/params.py
import hashlib, json

def params_hash(name: str, params: dict) -> str:
    payload = json.dumps({"name": name, "params": params}, sort_keys=True, ensure_ascii=False)
    return hashlib.sha1(payload.encode("utf-8")).hexdigest()
```

### IndicatorPipeline: 조합, 의존성, 증분 계산, 출력 표준화
```python
# indicators/pipeline.py
from __future__ import annotations
from dataclasses import dataclass
from typing import List, Dict, Any, Iterable
import pandas as pd
from indicators.registry import IndicatorRegistry
from indicators.base_indicator import IndicatorParams
from core.params import params_hash

@dataclass
class IndicatorSpec:
    name: str
    params: Dict[str, Any]
    save: bool = True

class IndicatorPipeline:
    def __init__(self, specs: Iterable[IndicatorSpec]):
        self.specs = list(specs)

    def warmup_days(self) -> int:
        """전체 파이프라인에서 필요한 최대 워밍업 일수."""
        max_warmup = 0
        for s in self.specs:
            cls = IndicatorRegistry.get(s.name)()
            max_warmup = max(max_warmup, cls.warmup(IndicatorParams(s.name, s.params)))
        return max_warmup

    def run(self, df: pd.DataFrame) -> Dict[str, pd.Series]:
        """입력 df는 최소한 date index와 가격 컬럼 보유. 반환은 {key: Series}."""
        out: Dict[str, pd.Series] = {}
        for s in self.specs:
            ind = IndicatorRegistry.get(s.name)()
            p = IndicatorParams(s.name, s.params)
            res = ind.compute(df, p)
            if isinstance(res, dict):
                for k, series in res.items():
                    out[k] = series
            else:
                out[res.name] = res
        return out

    def to_long_dataframe(self, symbol: str, market: str, source: str, date_index: pd.Index, outputs: Dict[str, pd.Series]) -> pd.DataFrame:
        rows = []
        for key, series in outputs.items():
            # key 예: "sma_20_close" or "macd_signal_12_26_9_close"
            name_parts = key.split("_")
            ind_name = name_parts[0]  # 대분류 이름
            # 파라미터는 spec에서 해시 계산: 동일 이름/파라미터 매칭
            spec_match = next((s for s in self.specs if s.name == ind_name), None)
            p_hash = params_hash(ind_name, spec_match.params if spec_match else {})
            df_tmp = series.to_frame("value")
            df_tmp["symbol"] = symbol
            df_tmp["market"] = market
            df_tmp["source"] = source
            df_tmp["indicator"] = key
            df_tmp["params_hash"] = p_hash
            df_tmp.reset_index(inplace=True)
            df_tmp.rename(columns={df_tmp.columns[0]: "date"}, inplace=True)
            rows.append(df_tmp[["symbol","date","indicator","params_hash","value","market","source"]])
        if not rows:
            return pd.DataFrame()
        return pd.concat(rows, axis=0, ignore_index=True)
```

설명:
- `warmup_days()`로 전체 파이프라인의 최대 필요 선행 데이터 길이를 산출 → 증분 계산 시 `latest_date - warmup`부터 재계산해 경계 왜곡 방지.
- `run()`은 여러 지표를 한 번에 계산, 복수 시리즈를 반환하는 지표(MACD, Bollinger)도 지원.
- `to_long_dataframe()`는 저장 표준 스키마로 변환.

### 저장 전략: Long/Wide/Hybrid
```python
# savers/indicator_saver.py
from __future__ import annotations
import pandas as pd
from sqlalchemy.engine import Engine
from core.db_manager import DBManager

class IndicatorSaver:
    def __init__(self, engine: Engine, mode: str = "long", table_long: str = "stock_indicators", wide_prefix: str = "ind_"):
        self.engine = engine
        self.mode = mode
        self.table_long = table_long
        self.wide_prefix = wide_prefix

    def save_long(self, df_long: pd.DataFrame) -> int:
        if df_long is None or df_long.empty:
            return 0
        return DBManager.upsert_dataframe(self.engine, df_long, table=self.table_long, unique_keys=("symbol","date","indicator","params_hash"))

    def merge_wide(self, price_df: pd.DataFrame, indicator_outputs: dict) -> pd.DataFrame:
        # price_df: columns include symbol/date; index agnostic
        merged = price_df.copy()
        for key, s in indicator_outputs.items():
            col = f"{self.wide_prefix}{key}"
            merged[col] = s.reindex(index=merged.index) if merged.index.equals(s.index) else s.values
        return merged
```

- `mode=long`이면 `to_long_dataframe()`의 결과를 `stock_indicators`에 업서트.
- `mode=wide`이면 저장 직전 가격 프레임에 컬럼 병합 후 기존 가격 테이블에 업서트.

### 수집-지표-저장 파이프라인 연결 예시(KR 주식, Daily Update)
```python
# scripts/daily_update.py (확장 예시)
from __future__ import annotations
import yaml, pandas as pd
from core.date_utils import DateUtils
from core.db_manager import DBManager, DBConfig
from collectors.kr_stock import KRStockCollector
from indicators.pipeline import IndicatorPipeline, IndicatorSpec
from savers.indicator_saver import IndicatorSaver


def build_pipeline(cfg) -> IndicatorPipeline:
    specs = [IndicatorSpec(**x) for x in cfg["indicators"]["pipeline"]]
    return IndicatorPipeline(specs)


def main():
    cfg = yaml.safe_load(open("config/settings.yaml"))
    db_cfg = DBConfig(**cfg["database"])
    engine = DBManager.get_engine(db_cfg)

    du = DateUtils(cfg["runtime"]["timezone"])
    if not du.after_market_close_with_buffer("XKRX", cfg["markets"]["close_times"]["XKRX"], cfg["markets"]["safe_delay_minutes"]):
        print("Not time yet. Exiting.")
        return

    symbols = ["005930", "000660"]
    collector = KRStockCollector(engine)

    pipe = build_pipeline(cfg)
    saver = IndicatorSaver(
        engine,
        mode=cfg["indicators"]["materialization"]["mode"],
        table_long=cfg["indicators"]["materialization"].get("table_long", "stock_indicators"),
        wide_prefix=cfg["indicators"]["materialization"].get("attach_to_price", {}).get("prefix", "ind_")
    )

    last_biz = du.get_last_business_day("XKRX")
    warmup = pipe.warmup_days()

    for s in symbols:
        latest = DBManager.get_latest_date(engine, collector.table, s)
        # warmup 보정: 최신일자에서 warmup 일수만큼 되돌아가 재계산
        start = last_biz if not latest else pd.to_datetime(latest) - pd.Timedelta(days=warmup)
        df_price = collector.fetch(s, start=start.date(), end=last_biz)
        df_price = collector.validate(df_price)
        if df_price is None or df_price.empty:
            continue
        # index를 date로 정규화(지표 계산 편의)
        dfp = df_price.copy()
        dfp["date"] = pd.to_datetime(dfp["date"])
        dfp.set_index("date", inplace=True)

        outputs = pipe.run(dfp)

        if saver.mode in ("long", "hybrid"):
            df_long = pipe.to_long_dataframe(symbol=s, market="XKRX", source="FDR", date_index=dfp.index, outputs=outputs)
            saver.save_long(df_long)

        if saver.mode in ("wide", "hybrid"):
            merged = saver.merge_wide(dfp, outputs)
            # wide 저장 시, 다시 date를 컬럼으로 복원하고 필요한 컬럼만 선택
            merged = merged.reset_index()
            merged["symbol"] = s
            merged["market"] = "XKRX"
            merged["source"] = "FDR"
            DBManager.upsert_dataframe(engine, merged, table=collector.table)

if __name__ == "__main__":
    main()
```

### 증분 처리 전략(성능/정합성)
- 워밍업 재계산: `latest_date - warmup_days()`부터 다시 계산하여 경계 값 왜곡 방지(EMA/RSI 등).
- 지표 전용 테이블은 `PRIMARY KEY(symbol, date, indicator, params_hash)`로 업서트 멱등 보장.
- 멀티스레딩: 심볼 단위 병렬 처리 + 레이트 리밋 유지. 파이프라인 계산은 CPU 바운드 → 워커 수 조정.
- 배치 업서트: 심볼별 계산 후 1천~5천 행 단위로 모아서 `upsert_dataframe` 호출.

### 유효성 검증/품질 관리
- 입력 컬럼 존재 여부: 각 Indicator의 `required_columns()`로 사전 점검.
- 값 검증: NaN 비율이 높은 시리즈는 경고 및 저장 제외 옵션.
- 경계 조건: `Low <= High`, `Open/Close` 범위 체크는 기존 `validate()` 유지.
- 메타데이터: 지표 계산 시작/종료 시간, 대상 건수, 실패 건수, 파라미터 해시를 구조화 로그로 남김.

---

### 테스트 전략
- 유닛: 각 Indicator에 대해 고정 샘플 시계열로 기대값 비교(예: pandas_ta/TA-Lib 참조값과의 오차 허용범위 테스트).
- 통합: 수집→파이프라인→저장 end-to-end 테스트(임시 MySQL 스키마).
- 회귀: 파라미터 변경 시 `params_hash`가 바뀌어 기존 값과 공존하는지 확인.

---

### 운영 팁
- 지표 수가 많아지면 long form 추천. 분석 시 `WHERE indicator IN (...) AND params_hash IN (...)`로 필터링.
- 인기 지표(예: `sma_20_close`, `rsi_14_close`)만 wide로 병합하는 `hybrid` 모드가 실무에 유리.
- 캐시: `feature_store.py`에 심볼별 최근 N일 price/indicator를 로컬 캐시해 당일 재실행 비용 절감.
- 버전 관리: 지표 알고리즘 변경 시 `name`에 버전 접미사(예: `rsi_v2`) 또는 `params`에 `algo_version`을 추가해 `params_hash`로 구분.
